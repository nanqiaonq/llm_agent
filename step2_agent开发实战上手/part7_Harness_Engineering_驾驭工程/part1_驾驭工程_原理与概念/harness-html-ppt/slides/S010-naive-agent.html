# 目的：演示没有 Harness 保护的 tool-calling 循环如何失效

import os, json, subprocess
from openai import OpenAI

MODEL = os.getenv("OPENAI_MODEL", "gpt-5")
SAFE_MAX_ITER = 3  # 课堂安全限制
client = OpenAI()

tools = [...]  # read_file / write_file / run_pytest

def read_file(path):
    # 故障④：FileNotFoundError 直接往上抛，被主循环 except 吞掉
    with open(path) as f: return f.read()

def write_file(path, content):
    # 故障⑥：没有权限闸，agent 可直接覆盖任何文件
    with open(path, "w") as f: f.write(content)
    return "done"

def run_pytest():
    result = subprocess.run(["pytest"], capture_output=True, text=True)
    # 故障④：returncode 被丢弃，agent 看不到失败信号
    return result.stdout

def naive_agent(task):
    messages = [
        # 故障③：system prompt 嵌入变化计数器，每轮破坏前缀
        {"role": "system", "content": "你是编程助手。[session_iter=0]"},
        {"role": "user", "content": task}
    ]
    # 故障①：真正的 naive agent 这里是 while True
    for iteration in range(SAFE_MAX_ITER):
        messages[0]["content"] = f"你是编程助手。[session_iter={iteration}]"
        response = client.chat.completions.create(
            model=MODEL, messages=messages, tools=tools
        )
        msg = response.choices[0].message
        messages.append(msg)  # 故障②：messages 无限膨胀

        if msg.tool_calls:
            for tc in msg.tool_calls:
                fn_name = tc.function.name
                fn_args = json.loads(tc.function.arguments)
                try:
                    if fn_name == "read_file": result = read_file(**fn_args)
                    elif fn_name == "write_file": result = write_file(**fn_args)
                    elif fn_name == "run_pytest": result = run_pytest()
                except Exception:
                    result = ""  # 故障④：异常被吞成空字符串
                messages.append({"role": "tool", "tool_call_id": tc.id, "content": result})
        else:
            break  # 故障⑦：agent 自己说完成就退出，无独立评审
    # 故障⑤：退出即丢失，下次重启从零开始
    # 故障⑧：没有任何 cost budget

naive_agent("读取 test_calculator.py，修复失败测试，运行 pytest")
